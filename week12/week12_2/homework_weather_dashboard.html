Weather dashboard

The Friday homework: a weather widget on the free Open-Meteo API. Pick a place by city name, by latitude and longitude, or with the browser’s location, and an effect fetches the current temperature, wind and conditions.

The brief

Open-Meteo API: free, no API key required.

Input: latitude + longitude OR city name (geocoding endpoint).

Display: current temperature, wind speed, weather description.

useEffect fetch: on coordinate change; show a loading state; handle errors (network down, bad coords, rate limit).

Bonus: auto-detect the user location with navigator.geolocation.

Submit by Sunday: GitHub repo URL + live GitHub Pages URL. Also: SprintQuiz.html. Implementing react elements within personal page.

The component

WeatherDashboard.jsx

const WEATHER_CODES = {
  0: 'Clear sky', 1: 'Mainly clear', 2: 'Partly cloudy', 3: 'Overcast',
  45: 'Fog', 48: 'Rime fog',
  51: 'Light drizzle', 53: 'Drizzle', 55: 'Dense drizzle',
  56: 'Freezing drizzle', 57: 'Dense freezing drizzle',
  61: 'Light rain', 63: 'Rain', 65: 'Heavy rain',
  66: 'Freezing rain', 67: 'Heavy freezing rain',
  71: 'Light snow', 73: 'Snow', 75: 'Heavy snow', 77: 'Snow grains',
  80: 'Light showers', 81: 'Showers', 82: 'Violent showers',
  85: 'Snow showers', 86: 'Heavy snow showers',
  95: 'Thunderstorm', 96: 'Thunderstorm with hail', 99: 'Thunderstorm with heavy hail',
};

function describe(code) {
  return WEATHER_CODES[code] || 'Unknown (code ' + code + ')';
}

function errorMessage(err, status) {
  if (status === 429) return 'Too many requests (rate limit). Wait a minute and try again.';
  if (status === 400) return 'Those coordinates were rejected. Latitude must be -90 to 90, longitude -180 to 180.';
  if (err instanceof TypeError) return 'Network error: check your connection.';
  return err.message;
}

function WeatherDashboard() {
  const [place, setPlace]     = useState({ name: 'Seoul', lat: 37.57, lon: 126.98 });
  const [city, setCity]       = useState('');
  const [weather, setWeather] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError]     = useState(null);
  const [locating, setLocating] = useState(false);

  // Fetch the weather whenever the coordinates change
  useEffect(() => {
    const controller = new AbortController();
    setLoading(true); setError(null);
    let status = 0;

    const url = 'https://api.open-meteo.com/v1/forecast'
      + `?latitude=${place.lat}&longitude=${place.lon}`
      + '&current=temperature_2m,wind_speed_10m,weather_code';

    fetch(url, { signal: controller.signal })
      .then(res => {
        status = res.status;
        if (!res.ok) throw new Error('Could not load weather (HTTP ' + res.status + ')');
        return res.json();
      })
      .then(data => { setWeather(data.current); setLoading(false); })
      .catch(err => {
        if (err.name === 'AbortError') return;
        setError(errorMessage(err, status)); setLoading(false);
      });

    return () => controller.abort();
  }, [place.lat, place.lon]);

  // City name -> coordinates (Open-Meteo geocoding endpoint)
  async function searchCity(e) {
    e.preventDefault();
    const name = city.trim();
    if (!name) return;
    setError(null);
    try {
      const res = await fetch('https://geocoding-api.open-meteo.com/v1/search?count=1&name='
                              + encodeURIComponent(name));
      if (!res.ok) throw new Error(errorMessage(new Error('Geocoding failed'), res.status));
      const data = await res.json();
      if (!data.results || data.results.length === 0) {
        setError('No place called “' + name + '” was found.');
        return;
      }
      const r = data.results[0];
      setPlace({ name: r.name + (r.country ? ', ' + r.country : ''), lat: r.latitude, lon: r.longitude });
    } catch (err) {
      setError(err instanceof TypeError ? errorMessage(err, 0) : err.message);
    }
  }

  // Latitude / longitude typed directly
  function useCoordinates(e) {
    e.preventDefault();
    const form = new FormData(e.target);
    const lat = Number(form.get('lat'));
    const lon = Number(form.get('lon'));
    if (!Number.isFinite(lat) || !Number.isFinite(lon) || Math.abs(lat) > 90 || Math.abs(lon) > 180) {
      setError('Latitude must be -90 to 90 and longitude -180 to 180.');
      return;
    }
    setPlace({ name: lat.toFixed(2) + ', ' + lon.toFixed(2), lat, lon });
  }

  // BONUS: the browser's own location
  function locate() {
    if (!('geolocation' in navigator)) {
      setError('This browser cannot share its location.');
      return;
    }
    setLocating(true);
    navigator.geolocation.getCurrentPosition(
      pos => {
        setLocating(false);
        const { latitude, longitude } = pos.coords;
        setPlace({ name: 'Your location', lat: latitude, lon: longitude });
      },
      err => {
        setLocating(false);
        setError('Location unavailable: ' + err.message);
      },
      { timeout: 10000 }
    );
  }

  return (
    <div>
      <form onSubmit={searchCity}>
        <label htmlFor="wx-city">City</label>{' '}
        <input id="wx-city" value={city} onChange={e => setCity(e.target.value)} placeholder="e.g. Busan" />
        <button>Search</button>
      </form>

      <form onSubmit={useCoordinates} key={place.lat + ',' + place.lon}>
        <label htmlFor="wx-lat">Latitude</label>{' '}
        <input id="wx-lat" name="lat" type="number" step="any" defaultValue={place.lat} style={{ width: '8rem' }} />{' '}
        <label htmlFor="wx-lon">Longitude</label>{' '}
        <input id="wx-lon" name="lon" type="number" step="any" defaultValue={place.lon} style={{ width: '8rem' }} />
        <button>Use coordinates</button>
      </form>

      <p>
        <button onClick={locate} disabled={locating}>
          {locating ? 'Locating…' : 'Use my location'}
        </button>
      </p>

      {error && <p className="error" role="alert">{error}</p>}

      {loading
        ? <p className="loading" role="status">Loading weather for {place.name}…</p>
        : weather && (
          <div className="weather" aria-live="polite">
            <p className="place">{place.name}</p>
            <p className="temp">{weather.temperature_2m} °C</p>
            <p>{describe(weather.weather_code)}</p>
            <p className="muted">wind {weather.wind_speed_10m} km/h</p>
          </div>
        )}
    </div>
  );
}

Against the brief

  • Open-Meteo, no key. Forecast from api.open-meteo.com/v1/forecast with current=temperature_2m,wind_speed_10m,weather_code, the same query as the lecturer’s starter.
  • Input: city name or coordinates. A city goes through geocoding-api.open-meteo.com/v1/search to get coordinates; the latitude and longitude fields set them directly.
  • Display. Temperature, wind speed, and weather_code turned into words with the WMO code table (the starter’s TODO 2).
  • useEffect on coordinate change. The effect depends on [place.lat, place.lon], shows a loading line while it runs, and aborts the previous request in its cleanup.
  • Errors. Network down (a TypeError from fetch), bad coordinates (checked before fetching, and HTTP 400 from the API), rate limit (HTTP 429), and a city that is not found.
  • Bonus. “Use my location” calls navigator.geolocation.getCurrentPosition; a refusal or timeout shows as an error, not a blank widget.
