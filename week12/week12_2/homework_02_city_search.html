The code

CityWeather.jsx

function CityWeather() {
  const [input, setInput]     = useState('Seoul');   // what is typed
  const [city, setCity]       = useState('Seoul');   // set when Search is pressed
  const [places, setPlaces]   = useState([]);
  const [place, setPlace]     = useState(null);      // { name, country, latitude, longitude, ... }
  const [weather, setWeather] = useState(null);
  const [searching, setSearching] = useState(true);
  const [loading, setLoading] = useState(false);
  const [error, setError]     = useState(null);

  // Effect 1: city name -> list of matching places
  useEffect(() => {
    if (!city) return;
    const controller = new AbortController();
    setSearching(true); setError(null); setPlaces([]);

    const url = 'https://geocoding-api.open-meteo.com/v1/search?count=5&language=en&format=json'
      + '&name=' + encodeURIComponent(city);

    fetch(url, { signal: controller.signal })
      .then(res => {
        if (!res.ok) throw new Error('City search failed (HTTP ' + res.status + ')');
        return res.json();
      })
      .then(data => {
        const found = data.results || [];   // no "results" key when nothing matches
        setSearching(false);
        if (found.length === 0) {
          setError('No city called “' + city + '” was found.');
          return;
        }
        setPlaces(found);
        setPlace(found[0]);                 // best match first
      })
      .catch(err => {
        if (err.name === 'AbortError') return;
        setError(err.message); setSearching(false);
      });

    return () => controller.abort();
  }, [city]);

  // Effect 2: chosen place -> current weather
  useEffect(() => {
    if (!place) return;
    const controller = new AbortController();
    setLoading(true); setError(null);

    const url = 'https://api.open-meteo.com/v1/forecast'
      + `?latitude=${place.latitude}&longitude=${place.longitude}`
      + '&current=temperature_2m,wind_speed_10m,weather_code';

    fetch(url, { signal: controller.signal })
      .then(res => {
        if (!res.ok) throw new Error('Could not load weather (HTTP ' + res.status + ')');
        return res.json();
      })
      .then(data => { setWeather(data.current); setLoading(false); })
      .catch(err => {
        if (err.name === 'AbortError') return;
        setWeather(null); setError(err.message); setLoading(false);
      });

    return () => controller.abort();
  }, [place]);

  function search(e) {
    e.preventDefault();
    if (input.trim()) setCity(input.trim());
  }

  return (
    <div>
      <form onSubmit={search}>
        <label htmlFor="hw2-city">City</label>{' '}
        <input id="hw2-city" value={input} onChange={e => setInput(e.target.value)}
               placeholder="e.g. Busan" autoComplete="off" />
        <button>Search</button>
      </form>

      {places.length > 1 && (
        <ul aria-label="Matching places">
          {places.map(p => (
            <li key={p.id}>
              <span>{p.name}, {p.admin1 ? p.admin1 + ', ' : ''}{p.country_code}</span>
              <button onClick={() => setPlace(p)} disabled={p.id === place.id}>
                {p.id === place.id ? 'Showing' : 'Show'}
              </button>
            </li>
          ))}
        </ul>
      )}

      {searching && <p className="loading" role="status">Searching for “{city}”…</p>}
      {error && <p className="error" role="alert">{error}</p>}
      {place && loading && <p className="loading" role="status">Loading weather for {place.name}…</p>}

      {place && weather && !loading && (
        <div className="weather" aria-live="polite">
          <p className="place">{place.name}, {place.country}</p>
          <p className="temp">{weather.temperature_2m} °C</p>
          <p>wind {weather.wind_speed_10m} km/h · code {weather.weather_code}</p>
          <p className="muted">({place.latitude.toFixed(2)}, {place.longitude.toFixed(2)})</p>
        </div>
      )}
    </div>
  );
}

Notes

  • Two effects, one job each. The first runs when the search term changes and finds places. The second runs when the chosen place changes and loads its weather. Both abort their request in the cleanup.
  • Typing is not searching. input follows the text box; city only changes on submit. If the effect depended on input, every keystroke would call the geocoding API.
  • No match means no results key. The API returns an object without results for an unknown name, so the code reads data.results || [] and shows “No city called ... was found.” Try asdfgh.
  • Several places share a name. The request asks for count=5, the best match is shown first, and the list lets you pick another one, such as a different Busan.
  • Search state is separate from weather state. searching and loading are two flags because the two requests can be in flight at different times.
  • The place keeps its coordinates. The weather request uses place.latitude and place.longitude straight from the geocoding result; the user never types them in this step.
