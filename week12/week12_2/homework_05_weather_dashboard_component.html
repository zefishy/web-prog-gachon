Weather dashboard, step 5: WeatherDashboard component

The finished dashboard. It starts from your location (or Seoul), searches any city or coordinates, describes the weather in words and adds a 4-day forecast. Every request is aborted in its cleanup.

The step

1Starter 2City search 3Weather code descriptions 4Geolocation (bonus) 5WeatherDashboard component

Open-Meteo API: free, no API key required.

Input: latitude + longitude OR city name (geocoding endpoint).

Display: current temperature, wind speed, weather description.

useEffect fetch: on coordinate change; show a loading state; handle errors (network down, bad coords, rate limit).

Bonus: auto-detect the user location with navigator.geolocation.

Week 12 Friday slides — Homework: Weather Dashboard

Against the brief

  1. Step 1, the starter. Weather for the current coordinates, with loading and error states. The coordinates form is the starter’s latitude and longitude inputs.
  2. Step 2, TODO 1. A city name goes through the geocoding endpoint, and the list of matches lets you pick a different place.
  3. Step 3, TODO 2. describe(code) turns weather_code into words and an icon, for today and for each forecast day.
  4. Step 4, TODO 3 (bonus). navigator.geolocation supplies the first coordinates, with Seoul as the fallback.
  5. Extra. A 4-day forecast, humidity, feels-like temperature, and an aborted request in every cleanup.

const WEATHER_CODES = {
  0: ['Clear sky', '☀️'], 1: ['Mainly clear', '🌤️'], 2: ['Partly cloudy', '⛅'], 3: ['Overcast', '☁️'],
  45: ['Fog', '🌫️'], 48: ['Depositing rime fog', '🌫️'],
  51: ['Light drizzle', '🌦️'], 53: ['Moderate drizzle', '🌦️'], 55: ['Dense drizzle', '🌧️'],
  56: ['Light freezing drizzle', '🌧️'], 57: ['Dense freezing drizzle', '🌧️'],
  61: ['Slight rain', '🌦️'], 63: ['Moderate rain', '🌧️'], 65: ['Heavy rain', '🌧️'],
  66: ['Light freezing rain', '🌧️'], 67: ['Heavy freezing rain', '🌧️'],
  71: ['Slight snowfall', '🌨️'], 73: ['Moderate snowfall', '🌨️'], 75: ['Heavy snowfall', '❄️'], 77: ['Snow grains', '🌨️'],
  80: ['Slight rain showers', '🌦️'], 81: ['Moderate rain showers', '🌧️'], 82: ['Violent rain showers', '⛈️'],
  85: ['Slight snow showers', '🌨️'], 86: ['Heavy snow showers', '❄️'],
  95: ['Thunderstorm', '⛈️'], 96: ['Thunderstorm with slight hail', '⛈️'], 99: ['Thunderstorm with heavy hail', '⛈️']
};

const describe = code => WEATHER_CODES[code] || ['Unknown weather (code ' + code + ')', '❔'];

const SEOUL = { name: 'Seoul', country: 'South Korea', latitude: 37.57, longitude: 126.98 };

const round = n => Math.round(n * 100) / 100;
const dayName = iso => new Date(iso + 'T12:00:00')
  .toLocaleDateString('en-GB', { weekday: 'short', day: 'numeric', month: 'short' });

function errorMessage(err, status) {
  if (status === 429) return 'Too many requests (rate limit). Wait a minute and try again.';
  if (err instanceof TypeError) return 'Network error: check your connection.';
  return err.message;
}

function CityForm({ onSearch }) {
  const [input, setInput] = useState('');
  function submit(e) {
    e.preventDefault();
    if (input.trim()) onSearch(input.trim());
  }
  return (
    <form onSubmit={submit}>
      <label htmlFor="hw5-city">City</label>{' '}
      <input id="hw5-city" value={input} onChange={e => setInput(e.target.value)}
             placeholder="e.g. Jeju" autoComplete="off" />
      <button>Search</button>
    </form>
  );
}

function CoordinatesForm({ place, onUse }) {
  function submit(e) {
    e.preventDefault();
    const form = new FormData(e.target);
    onUse(Number(form.get('lat')), Number(form.get('lon')));
  }
  return (
    <form onSubmit={submit} key={place.latitude + ',' + place.longitude}>
      <label htmlFor="hw5-lat">Latitude</label>{' '}
      <input id="hw5-lat" name="lat" type="number" step="any" defaultValue={round(place.latitude)} style={{ width: '8rem' }} />{' '}
      <label htmlFor="hw5-lon">Longitude</label>{' '}
      <input id="hw5-lon" name="lon" type="number" step="any" defaultValue={round(place.longitude)} style={{ width: '8rem' }} />
      <button>Use coordinates</button>
    </form>
  );
}

function CurrentWeather({ place, current, units }) {
  const [text, icon] = describe(current.weather_code);
  return (
    <div className="weather" aria-live="polite">
      <p className="icon" aria-hidden="true">{icon}</p>
      <p className="place">{place.name}{place.country ? ', ' + place.country : ''}</p>
      <p className="temp">{Math.round(current.temperature_2m)}{units.temperature_2m}</p>
      <p>{text}</p>
      <p className="muted">
        feels like {Math.round(current.apparent_temperature)}{units.apparent_temperature}
        {' · '}humidity {current.relative_humidity_2m}%
        {' · '}wind {current.wind_speed_10m} km/h
      </p>
    </div>
  );
}

function Forecast({ daily }) {
  return (
    <section className="forecast" aria-label="4-day forecast">
      {daily.time.map((day, i) => {
        const [text, icon] = describe(daily.weather_code[i]);
        return (
          <div className="day" key={day}>
            <p><strong>{dayName(day)}</strong></p>
            <p className="icon" aria-hidden="true">{icon}</p>
            <p>{Math.round(daily.temperature_2m_max[i])}° / {Math.round(daily.temperature_2m_min[i])}°</p>
            <p className="muted">{text}</p>
          </div>
        );
      })}
    </section>
  );
}

function WeatherDashboard() {
  const [place, setPlace]     = useState(null);
  const [note, setNote]       = useState('Asking for your location…');
  const [city, setCity]       = useState('');
  const [matches, setMatches] = useState([]);
  const [data, setData]       = useState(null);
  const [loading, setLoading] = useState(false);
  const [error, setError]     = useState(null);

  // 1. On mount: the browser's location, otherwise Seoul
  useEffect(() => {
    let cancelled = false;
    if (!('geolocation' in navigator)) {
      setPlace(SEOUL); setNote('This browser cannot share its location, so Seoul is shown.');
      return;
    }
    navigator.geolocation.getCurrentPosition(
      pos => {
        if (cancelled) return;
        setPlace({ name: 'Your location', latitude: pos.coords.latitude, longitude: pos.coords.longitude });
        setNote('');
      },
      () => {
        if (cancelled) return;
        setPlace(SEOUL); setNote('Location not shared, so Seoul is shown. Search for any city above.');
      },
      { timeout: 8000 }
    );
    return () => { cancelled = true; };
  }, []);

  // 2. When a city is searched: look up matching places
  useEffect(() => {
    if (!city) return;
    const controller = new AbortController();
    setError(null);
    let status = 0;

    const url = 'https://geocoding-api.open-meteo.com/v1/search?count=5&language=en&format=json'
      + '&name=' + encodeURIComponent(city);

    fetch(url, { signal: controller.signal })
      .then(res => {
        status = res.status;
        if (!res.ok) throw new Error('City search failed (HTTP ' + res.status + ')');
        return res.json();
      })
      .then(json => {
        const found = json.results || [];
        setMatches(found);
        if (found.length === 0) { setError('No city called “' + city + '” was found.'); return; }
        setPlace(found[0]); setNote('');
      })
      .catch(err => {
        if (err.name === 'AbortError') return;
        setError(errorMessage(err, status));
      });

    return () => controller.abort();
  }, [city]);

  // 3. When the place changes: current weather + 4-day forecast
  useEffect(() => {
    if (!place) return;
    const controller = new AbortController();
    setLoading(true); setError(null);
    let status = 0;

    const url = 'https://api.open-meteo.com/v1/forecast'
      + `?latitude=${place.latitude}&longitude=${place.longitude}`
      + '&current=temperature_2m,apparent_temperature,relative_humidity_2m,wind_speed_10m,weather_code'
      + '&daily=weather_code,temperature_2m_max,temperature_2m_min&forecast_days=4&timezone=auto';

    fetch(url, { signal: controller.signal })
      .then(res => {
        status = res.status;
        if (res.status === 400) throw new Error('Those coordinates were rejected. Latitude must be -90 to 90, longitude -180 to 180.');
        if (!res.ok) throw new Error('Could not load weather (HTTP ' + res.status + ')');
        return res.json();
      })
      .then(json => { setData(json); setLoading(false); })
      .catch(err => {
        if (err.name === 'AbortError') return;
        setData(null); setError(errorMessage(err, status)); setLoading(false);
      });

    return () => controller.abort();
  }, [place]);

  function useCoordinates(lat, lon) {
    if (!Number.isFinite(lat) || !Number.isFinite(lon) || Math.abs(lat) > 90 || Math.abs(lon) > 180) {
      setError('Latitude must be -90 to 90 and longitude -180 to 180.');
      return;
    }
    setMatches([]); setNote('');
    setPlace({ name: lat.toFixed(2) + ', ' + lon.toFixed(2), latitude: lat, longitude: lon });
  }

  return (
    <div>
      <CityForm onSearch={setCity} />
      {place && <CoordinatesForm place={place} onUse={useCoordinates} />}

      {matches.length > 1 && (
        <ul aria-label="Matching places">
          {matches.map(m => (
            <li key={m.id}>
              <span>{m.name}, {m.admin1 ? m.admin1 + ', ' : ''}{m.country_code}</span>
              <button onClick={() => setPlace(m)} disabled={m.id === place.id}>
                {m.id === place.id ? 'Showing' : 'Show'}
              </button>
            </li>
          ))}
        </ul>
      )}

      {note && <p className="muted" role="status">{note}</p>}
      {error && <p className="error" role="alert">{error}</p>}
      {place && loading && <p className="loading" role="status">Loading weather for {place.name}…</p>}

      {place && data && !loading && (
        <div>
          <CurrentWeather place={place} current={data.current} units={data.current_units} />
          <Forecast daily={data.daily} />
        </div>
      )}
    </div>
  );
}

Notes

  • Three effects, three triggers. One on mount ([]) for the location, one on [city] for geocoding, one on [place] for the weather. Each has one dependency list and one cleanup.
  • Everything funnels into place. Geolocation, a city match, a match picked from the list and typed coordinates all call setPlace. Only the weather effect fetches weather, so there is one code path for it.
  • Child components take props only. CurrentWeather and Forecast have no state; CityForm keeps only its own text box and reports the finished search upward with onSearch.
  • Errors are named. A TypeError from fetch is a network failure, HTTP 429 is the rate limit, HTTP 400 is a rejected coordinate, and a city with no match gets its own message. Coordinates typed by hand are also range-checked before any request.
  • The coordinates form resets itself. Its key is the place’s coordinates, so choosing a new place remounts the form and the uncontrolled inputs pick up the new defaultValue.
  • Failure never leaves old data behind. A failed weather request calls setData(null), so the dashboard shows the error instead of the previous place’s numbers.
