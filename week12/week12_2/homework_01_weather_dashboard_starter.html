The code

WeatherStarter.jsx

// Homework starter: Weather Dashboard (Open-Meteo, no API key)
function WeatherStarter() {
  const [lat, setLat]         = useState(37.46);
  const [lon, setLon]         = useState(127.01);
  const [weather, setWeather] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError]     = useState(null);

  useEffect(() => {
    const controller = new AbortController();
    setLoading(true); setError(null);

    const url = 'https://api.open-meteo.com/v1/forecast'
      + `?latitude=${lat}&longitude=${lon}`
      + '&current=temperature_2m,wind_speed_10m,weather_code';

    fetch(url, { signal: controller.signal })
      .then(res => {
        if (!res.ok) throw new Error('Could not load weather (check coordinates)');
        return res.json();
      })
      .then(data => { setWeather(data.current); setLoading(false); })
      .catch(err => {
        if (err.name === 'AbortError') return;
        setWeather(null); setError(err.message); setLoading(false);
      });

    return () => controller.abort();
  }, [lat, lon]);

  return (
    <div>
      <div>
        <label htmlFor="hw1-lat">Latitude</label>{' '}
        <input id="hw1-lat" type="number" step="any" value={lat}
               onChange={e => setLat(e.target.value)} style={{ width: '8rem' }} />{' '}
        <label htmlFor="hw1-lon">Longitude</label>{' '}
        <input id="hw1-lon" type="number" step="any" value={lon}
               onChange={e => setLon(e.target.value)} style={{ width: '8rem' }} />
      </div>

      {loading && <p className="loading" role="status">Loading weather…</p>}
      {error && <p className="error" role="alert">{error}</p>}
      {weather && !loading && (
        <div className="weather" aria-live="polite">
          <p className="temp">{weather.temperature_2m} °C</p>
          <p>wind {weather.wind_speed_10m} km/h · code {weather.weather_code}</p>
        </div>
      )}
    </div>
  );
}

Notes

  • The effect depends on [lat, lon]. Every keystroke in either field changes a dependency, so the effect runs again and fetches the new place.
  • Three states for three outcomes. loading covers the wait, error a failed request, and weather the data. The weather line only renders when weather && !loading.
  • The cleanup aborts the old request. Typing quickly starts several fetches; controller.abort() cancels the ones that are no longer wanted, and the AbortError they throw is ignored in the catch.
  • Why latitude 120 fails. Open-Meteo answers HTTP 400 for a latitude outside -90 to 90. fetch does not reject on a 400, so the code checks res.ok and throws its own error.
  • Clear stale data on failure. The error branch calls setWeather(null); otherwise the previous place’s numbers would sit under the next successful label.
  • The next three steps are the three TODOs in the starter comment: city search, weather code descriptions, and geolocation.
