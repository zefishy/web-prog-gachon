The code

WeatherWithDescription.jsx

const WEATHER_CODES = {
  0: ['Clear sky', '☀️'], 1: ['Mainly clear', '🌤️'], 2: ['Partly cloudy', '⛅'], 3: ['Overcast', '☁️'],
  45: ['Fog', '🌫️'], 48: ['Depositing rime fog', '🌫️'],
  51: ['Light drizzle', '🌦️'], 53: ['Moderate drizzle', '🌦️'], 55: ['Dense drizzle', '🌧️'],
  56: ['Light freezing drizzle', '🌧️'], 57: ['Dense freezing drizzle', '🌧️'],
  61: ['Slight rain', '🌦️'], 63: ['Moderate rain', '🌧️'], 65: ['Heavy rain', '🌧️'],
  66: ['Light freezing rain', '🌧️'], 67: ['Heavy freezing rain', '🌧️'],
  71: ['Slight snowfall', '🌨️'], 73: ['Moderate snowfall', '🌨️'], 75: ['Heavy snowfall', '❄️'], 77: ['Snow grains', '🌨️'],
  80: ['Slight rain showers', '🌦️'], 81: ['Moderate rain showers', '🌧️'], 82: ['Violent rain showers', '⛈️'],
  85: ['Slight snow showers', '🌨️'], 86: ['Heavy snow showers', '❄️'],
  95: ['Thunderstorm', '⛈️'], 96: ['Thunderstorm with slight hail', '⛈️'], 99: ['Thunderstorm with heavy hail', '⛈️']
};

const describe = code => WEATHER_CODES[code] || ['Unknown weather (code ' + code + ')', '❔'];

const PRESETS = [
  { name: 'Seoul',     latitude: 37.57, longitude: 126.98 },
  { name: 'Busan',     latitude: 35.18, longitude: 129.08 },
  { name: 'Singapore', latitude: 1.29,  longitude: 103.85 },
  { name: 'London',    latitude: 51.51, longitude: -0.13 },
  { name: 'Reykjavík', latitude: 64.15, longitude: -21.94 },
];

function WeatherWithDescription() {
  const [place, setPlace]     = useState(PRESETS[0]);
  const [weather, setWeather] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError]     = useState(null);

  useEffect(() => {
    const controller = new AbortController();
    setLoading(true); setError(null);

    const url = 'https://api.open-meteo.com/v1/forecast'
      + `?latitude=${place.latitude}&longitude=${place.longitude}`
      + '&current=temperature_2m,wind_speed_10m,weather_code';

    fetch(url, { signal: controller.signal })
      .then(res => {
        if (!res.ok) throw new Error('Could not load weather (HTTP ' + res.status + ')');
        return res.json();
      })
      .then(data => { setWeather(data.current); setLoading(false); })
      .catch(err => {
        if (err.name === 'AbortError') return;
        setWeather(null); setError(err.message); setLoading(false);
      });

    return () => controller.abort();
  }, [place]);

  // Derived while rendering: no state, no effect
  const [text, icon] = weather ? describe(weather.weather_code) : ['', ''];

  return (
    <div>
      <div role="group" aria-label="Place">
        {PRESETS.map(p => (
          <button key={p.name} onClick={() => setPlace(p)} aria-pressed={p === place}>{p.name}</button>
        ))}
      </div>

      {loading && <p className="loading" role="status">Loading weather for {place.name}…</p>}
      {error && <p className="error" role="alert">{error}</p>}
      {weather && !loading && (
        <div className="weather" aria-live="polite">
          <p className="icon" aria-hidden="true">{icon}</p>
          <p className="place">{place.name}</p>
          <p className="temp">{weather.temperature_2m} °C</p>
          <p>{text}</p>
          <p className="muted">wind {weather.wind_speed_10m} km/h · code {weather.weather_code}</p>
        </div>
      )}
    </div>
  );
}

Notes

  • A plain object is the lookup table. The keys are the WMO codes Open-Meteo uses, and each value is a [description, icon] pair that describe destructures.
  • Unknown codes do not crash. WEATHER_CODES[code] || [...] falls back to “Unknown weather (code N)” if the API ever adds a code the table lacks.
  • The description is derived, not stored. describe(weather.weather_code) runs during render. It needs no state and no effect, the same point as “You might not need an effect”.
  • Icons are decoration. The emoji sits in an element with aria-hidden="true"; the words carry the meaning for screen readers.
  • Table and demo share one idea, not one object. The table above is plain HTML listing the same 28 codes that the printed WEATHER_CODES holds.
  • Preset places, not city search. Five buttons stand in for the step 2 search so this step only changes one thing. The effect depends on [place] and refetches when a button is pressed.
