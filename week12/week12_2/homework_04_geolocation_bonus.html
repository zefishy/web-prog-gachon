const describe = code => WEATHER_CODES[code] || ['Unknown weather (code ' + code + ')', '❔'];

const SEOUL = { name: 'Seoul (default)', latitude: 37.57, longitude: 126.98 };

function LocalWeather() {
  const [coords, setCoords]   = useState(null);
  const [note, setNote]       = useState('Asking for your location…');
  const [weather, setWeather] = useState(null);
  const [loading, setLoading] = useState(true);
  const [error, setError]     = useState(null);

  // Effect 1 (on mount): the first coordinates come from the browser
  useEffect(() => {
    let cancelled = false;

    if (!('geolocation' in navigator)) {
      setCoords(SEOUL);
      setNote('This browser cannot share its location, so Seoul is shown.');
      return;
    }

    navigator.geolocation.getCurrentPosition(
      pos => {
        if (cancelled) return;
        setCoords({ name: 'Your location',
                    latitude: pos.coords.latitude,
                    longitude: pos.coords.longitude });
        setNote('Using your current location.');
      },
      err => {
        if (cancelled) return;
        setCoords(SEOUL);
        setNote('Location unavailable (' + err.message + '), so Seoul is shown.');
      },
      { timeout: 10000 }
    );

    return () => { cancelled = true; };   // ignore a late answer after unmount
  }, []);                                 // [] = ask once, on mount

  // Effect 2: coordinates -> weather
  useEffect(() => {
    if (!coords) return;
    const controller = new AbortController();
    setLoading(true); setError(null);

    const url = 'https://api.open-meteo.com/v1/forecast'
      + `?latitude=${coords.latitude}&longitude=${coords.longitude}`
      + '&current=temperature_2m,wind_speed_10m,weather_code';

    fetch(url, { signal: controller.signal })
      .then(res => {
        if (!res.ok) throw new Error('Could not load weather (HTTP ' + res.status + ')');
        return res.json();
      })
      .then(data => { setWeather(data.current); setLoading(false); })
      .catch(err => {
        if (err.name === 'AbortError') return;
        setWeather(null); setError(err.message); setLoading(false);
      });

    return () => controller.abort();
  }, [coords]);

  const [text, icon] = weather ? describe(weather.weather_code) : ['', ''];

  return (
    <div>
      <p className="muted" role="status">{note}</p>

      {loading && <p className="loading" role="status">Loading weather…</p>}
      {error && <p className="error" role="alert">{error}</p>}
      {coords && weather && !loading && (
        <div className="weather" aria-live="polite">
          <p className="icon" aria-hidden="true">{icon}</p>
          <p className="place">{coords.name}</p>
          <p className="temp">{weather.temperature_2m} °C</p>
          <p>{text}</p>
          <p className="muted">wind {weather.wind_speed_10m} km/h · ({coords.latitude.toFixed(2)}, {coords.longitude.toFixed(2)})</p>
        </div>
      )}
    </div>
  );
}

Notes

  • The browser asks first. Reload the page and a permission prompt appears. Allow it and the card shows “Your location”; block it, or let it time out, and the note says why Seoul is shown instead. Geolocation only works on HTTPS pages (GitHub Pages is HTTPS) or on localhost.
  • [] means once, on mount. Asking for the position is a one-time setup step, so the dependency array is empty.
  • The cleanup sets a flag. getCurrentPosition cannot be aborted, so cancelled = true makes a late answer do nothing if the component has already unmounted.
  • Every failure path still sets coordinates. No geolocation in navigator, permission denied, position unavailable and a timeout all end in setCoords(SEOUL). The weather effect never waits for coordinates that will not come.
  • A second effect chains off the first. The weather effect depends on [coords]. It returns early while coords is null, and runs once the first effect has set it.
  • Say what happened. The note state tells the user whether the numbers are for their position or for the fallback, so a refusal never looks like a bug.
