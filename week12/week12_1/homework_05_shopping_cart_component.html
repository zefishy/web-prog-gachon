Shopping cart, step 5: ShoppingCart component

The finished cart, split into components. ShoppingCart owns the state; ProductList and CartPanel only receive data as props and report clicks back through callbacks.

const PRODUCTS = [

  { id: 2, name: 'Mechanical Keyboard', price: 129000, image: 'img/keyboard.jpg' },
  { id: 3, name: 'USB-C Hub',           price: 45000,  image: 'img/usb-hub.jpg' },
  { id: 4, name: 'Laptop Stand',        price: 39000,  image: 'img/laptop-stand.jpg' },
  { id: 5, name: 'HD Webcam',           price: 72000,  image: 'img/webcam.jpg' },
];

const won = n => n.toLocaleString('ko-KR') + ' KRW';

function ProductList({ products, isInCart, onAdd, onRemove }) {
  return (
    <ul className="product-grid">
      {products.map(p => (
        <li key={p.id}>
          <img src={p.image} alt="" width="320" height="200" />
          <h3>{p.name}</h3>
          <p className="price">{won(p.price)}</p>
          {isInCart(p.id)
            ? <button onClick={() => onRemove(p.id)}>Remove</button>
            : <button onClick={() => onAdd(p)}>Add to cart</button>}
        </li>
      ))}
    </ul>
  );
}

function CartPanel({ cart, onIncrease, onDecrease }) {
  const total = cart.reduce((sum, item) => sum + item.price * item.qty, 0);
  const count = cart.reduce((sum, item) => sum + item.qty, 0);
  return (
    <aside className="cart-panel" aria-label="Cart">
      <h3>Cart ({count})</h3>
      {cart.length === 0
        ? <p className="muted">Your cart is empty.</p>
        : (
          <ul>
            {cart.map(item => (
              <li key={item.id}>
                <span className="line-name">{item.name}</span>
                <span className="qty">
                  <button onClick={() => onDecrease(item.id)} aria-label={'One fewer ' + item.name}>−</button>
                  <span aria-label={'Quantity of ' + item.name}>{item.qty}</span>
                  <button onClick={() => onIncrease(item.id)} aria-label={'One more ' + item.name}>+</button>
                </span>
                <span className="line-price">{won(item.price * item.qty)}</span>
              </li>
            ))}
          </ul>
        )}
      <p className="total" aria-live="polite"><span>Total</span><span>{won(total)}</span></p>
    </aside>
  );
}

function ShoppingCart({ products }) {
  const [cart, setCart] = useState([]);

  const isInCart = id => cart.some(item => item.id === id);
  const add = product => setCart(prev => [...prev, { ...product, qty: 1 }]);
  const remove = id => setCart(prev => prev.filter(item => item.id !== id));
  const increase = id => setCart(prev => prev.map(item =>
    item.id === id ? { ...item, qty: item.qty + 1 } : item
  ));
  const decrease = id => setCart(prev => prev
    .map(item => item.id === id ? { ...item, qty: item.qty - 1 } : item)
    .filter(item => item.qty > 0)
  );

  return (
    <div className="shop with-cart">
      <ProductList products={products} isInCart={isInCart}
                   onAdd={add} onRemove={remove} />
      <CartPanel cart={cart} onIncrease={increase} onDecrease={decrease} />
    </div>
  );
}

Notes

  • State lives in the nearest common parent. Both children need the cart, so it belongs to ShoppingCart: the lifting-state-up rule from Part 5 of the slides.
  • Data down, events up. ProductList gets isInCart, onAdd and onRemove; CartPanel gets the cart plus onIncrease and onDecrease.
  • Neither child has useState. They are plain functions of their props, which makes each one easy to read and reuse.
  • The behaviour is identical to step 4. This step changes the structure, not what the user sees.
