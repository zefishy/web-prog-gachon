Shopping cart, step 4: Quantity controls

Each cart line gets a quantity with + and − buttons. Changing one item’s quantity means building a new array with map and a new object for that one item.

The code

Shop.jsx

const PRODUCTS = [
  { id: 1, name: 'Wireless Earbuds',    price: 89000,  image: 'img/earbuds.jpg' },
  { id: 2, name: 'Mechanical Keyboard', price: 129000, image: 'img/keyboard.jpg' },
  { id: 3, name: 'USB-C Hub',           price: 45000,  image: 'img/usb-hub.jpg' },
  { id: 4, name: 'Laptop Stand',        price: 39000,  image: 'img/laptop-stand.jpg' },
  { id: 5, name: 'HD Webcam',           price: 72000,  image: 'img/webcam.jpg' },
];

const won = n => n.toLocaleString('ko-KR') + ' KRW';

function Shop() {
  const [cart, setCart] = useState([]);

  const lineOf = id => cart.find(item => item.id === id);

  const add = product => setCart(prev => [...prev, { ...product, qty: 1 }]);
  const remove = id => setCart(prev => prev.filter(item => item.id !== id));
  const increase = id => setCart(prev => prev.map(item =>
    item.id === id ? { ...item, qty: item.qty + 1 } : item
  ));
  const decrease = id => setCart(prev => prev
    .map(item => item.id === id ? { ...item, qty: item.qty - 1 } : item)
    .filter(item => item.qty > 0)
  );

  const total = cart.reduce((sum, item) => sum + item.price * item.qty, 0);
  const count = cart.reduce((sum, item) => sum + item.qty, 0);

  return (
    <div className="shop with-cart">
      <ul className="product-grid">
        {PRODUCTS.map(p => (
          <li key={p.id}>
            <img src={p.image} alt="" width="320" height="200" />
            <h3>{p.name}</h3>
            <p className="price">{won(p.price)}</p>
            {lineOf(p.id)
              ? <button onClick={() => remove(p.id)}>Remove</button>
              : <button onClick={() => add(p)}>Add to cart</button>}
          </li>
        ))}
      </ul>

      <aside className="cart-panel" aria-label="Cart">
        <h3>Cart ({count})</h3>
        {cart.length === 0
          ? <p className="muted">Your cart is empty.</p>
          : (
            <ul>
              {cart.map(item => (
                <li key={item.id}>
                  <span className="line-name">{item.name}</span>
                  <span className="qty">
                    <button onClick={() => decrease(item.id)} aria-label={'One fewer ' + item.name}>−</button>
                    <span aria-label={'Quantity of ' + item.name}>{item.qty}</span>
                    <button onClick={() => increase(item.id)} aria-label={'One more ' + item.name}>+</button>
                  </span>
                  <span className="line-price">{won(item.price * item.qty)}</span>
                </li>
              ))}
            </ul>
          )}
        <p className="total" aria-live="polite"><span>Total</span><span>{won(total)}</span></p>
      </aside>
    </div>
  );
}

Notes

  • Each cart item now carries qty, added with { ...product, qty: 1 } when it first goes in.
  • + uses the Toggle pattern from the slides: map copies every item, and the one that matches gets a new object with qty + 1.
  • − at 1 removes the line. The map is followed by a filter that drops anything at zero, so the cart never shows a quantity of 0.
  • Both the total and the cart count now multiply by quantity, and both are still derived with reduce.
