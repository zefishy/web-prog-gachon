Homework: Interactive shopping cart

The Monday homework as one page. Five sections build the same cart one requirement at a time, and section 5 is the finished ShoppingCart component. Each section has its own live demo and its own code.

The sections

1Product list 2Add / Remove buttons 3Cart sidebar 4Quantity controls 5ShoppingCart component

Every section works on its own: the cart in section 3 is separate from the one in section 5. The five sections share one product array, printed once under section 1.

1. Product list

1. Product list. At least 5 items: name, price, image URL.

const won = n => n.toLocaleString('ko-KR') + ' KRW';

function ProductList({ products }) {
  return (
    <ul className="product-grid">
      {products.map(p => (
        <li key={p.id}>
          <img src={p.image} alt="" width="320" height="200" />
          <h3>{p.name}</h3>
          <p className="price">{won(p.price)}</p>
        </li>
      ))}
    </ul>
  );
}

A plain function component. The array is data, map turns each object into a list item, and key gives React a stable identity for each row.

2. Add / Remove buttons

2. Add / Remove buttons. State: array of cart items.

Week 12 Monday slides — Homework: Interactive Shopping Cart

Uses PRODUCTS and won from section 1.

Shop.jsx

function Shop() {
  const [cart, setCart] = useState([]);

  const inCart = id => cart.some(item => item.id === id);
  const add    = product => setCart(prev => [...prev, product]);
  const remove = id => setCart(prev => prev.filter(item => item.id !== id));

  return (
    <div>
      <ul className="product-grid">
        {PRODUCTS.map(p => (
          <li key={p.id}>
            <img src={p.image} alt="" width="320" height="200" />
            <h3>{p.name}</h3>
            <p className="price">{won(p.price)}</p>
            {inCart(p.id)
              ? <button onClick={() => remove(p.id)}>Remove</button>
              : <button onClick={() => add(p)}>Add to cart</button>}
          </li>
        ))}
      </ul>
      <p aria-live="polite">Cart state: [{cart.map(item => item.name).join(', ')}]</p>
    </div>
  );
}

cart is an array in state. Adding and removing never mutate it: [...prev, product] and filter both return a new array, so React sees a change.

3. Cart sidebar

3. Cart sidebar. Items + total price, computed from state.

Uses PRODUCTS and won from section 1.

Shop.jsx

function Shop() {
  const [cart, setCart] = useState([]);

  const inCart = id => cart.some(item => item.id === id);
  const add    = product => setCart(prev => [...prev, product]);
  const remove = id => setCart(prev => prev.filter(item => item.id !== id));

  const total = cart.reduce((sum, item) => sum + item.price, 0);

  return (
    <div className="shop with-cart">
      <ul className="product-grid">
        {PRODUCTS.map(p => (
          <li key={p.id}>
            <img src={p.image} alt="" width="320" height="200" />
            <h3>{p.name}</h3>
            <p className="price">{won(p.price)}</p>
            {inCart(p.id)
              ? <button onClick={() => remove(p.id)}>Remove</button>
              : <button onClick={() => add(p)}>Add to cart</button>}
          </li>
        ))}
      </ul>

      <aside className="cart-panel" aria-label="Cart">
        <h3>Cart ({cart.length})</h3>
        {cart.length === 0
          ? <p className="muted">Your cart is empty.</p>
          : (
            <ul>
              {cart.map(item => (
                <li key={item.id}>
                  <span className="line-name">{item.name}</span>
                  <span className="line-price">{won(item.price)}</span>
                </li>
              ))}
            </ul>
          )}
        <p className="total" aria-live="polite"><span>Total</span><span>{won(total)}</span></p>
      </aside>
    </div>
  );
}

The total is not stored. It is calculated from cart on every render with reduce, so it can never disagree with the items.

4. Quantity controls

4. Quantity controls. + and - per item.

Week 12 Monday slides — Homework: Interactive Shopping Cart

Uses PRODUCTS and won from section 1.

Shop.jsx

function Shop() {
  const [cart, setCart] = useState([]);

  const lineOf = id => cart.find(item => item.id === id);

  const add = product => setCart(prev => [...prev, { ...product, qty: 1 }]);
  const remove = id => setCart(prev => prev.filter(item => item.id !== id));
  const increase = id => setCart(prev => prev.map(item =>
    item.id === id ? { ...item, qty: item.qty + 1 } : item
  ));
  const decrease = id => setCart(prev => prev
    .map(item => item.id === id ? { ...item, qty: item.qty - 1 } : item)
    .filter(item => item.qty > 0)
  );

  const total = cart.reduce((sum, item) => sum + item.price * item.qty, 0);
  const count = cart.reduce((sum, item) => sum + item.qty, 0);

  return (
    <div className="shop with-cart">
      <ul className="product-grid">
        {PRODUCTS.map(p => (
          <li key={p.id}>
            <img src={p.image} alt="" width="320" height="200" />
            <h3>{p.name}</h3>
            <p className="price">{won(p.price)}</p>
            {lineOf(p.id)
              ? <button onClick={() => remove(p.id)}>Remove</button>
              : <button onClick={() => add(p)}>Add to cart</button>}
          </li>
        ))}
      </ul>

      <aside className="cart-panel" aria-label="Cart">
        <h3>Cart ({count})</h3>
        {cart.length === 0
          ? <p className="muted">Your cart is empty.</p>
          : (
            <ul>
              {cart.map(item => (
                <li key={item.id}>
                  <span className="line-name">{item.name}</span>
                  <span className="qty">
                    <button onClick={() => decrease(item.id)} aria-label={'One fewer ' + item.name}>−</button>
                    <span aria-label={'Quantity of ' + item.name}>{item.qty}</span>
                    <button onClick={() => increase(item.id)} aria-label={'One more ' + item.name}>+</button>
                  </span>
                  <span className="line-price">{won(item.price * item.qty)}</span>
                </li>
              ))}
            </ul>
          )}
        <p className="total" aria-live="polite"><span>Total</span><span>{won(total)}</span></p>
      </aside>
    </div>
  );
}

Each cart line now has a qty. increase and decrease map over the array and replace one object; decrease also filters out lines that reach 0.

5. ShoppingCart component

5. ShoppingCart component. Product list + cart panel.

ShoppingCart.jsx

function ProductList({ products, isInCart, onAdd, onRemove }) {
  return (
    <ul className="product-grid">
      {products.map(p => (
        <li key={p.id}>
          <img src={p.image} alt="" width="320" height="200" />
          <h3>{p.name}</h3>
          <p className="price">{won(p.price)}</p>
          {isInCart(p.id)
            ? <button onClick={() => onRemove(p.id)}>Remove</button>
            : <button onClick={() => onAdd(p)}>Add to cart</button>}
        </li>
      ))}
    </ul>
  );
}

function CartPanel({ cart, onIncrease, onDecrease }) {
  const total = cart.reduce((sum, item) => sum + item.price * item.qty, 0);
  const count = cart.reduce((sum, item) => sum + item.qty, 0);
  return (
    <aside className="cart-panel" aria-label="Cart">
      <h3>Cart ({count})</h3>
      {cart.length === 0
        ? <p className="muted">Your cart is empty.</p>
        : (
          <ul>
            {cart.map(item => (
              <li key={item.id}>
                <span className="line-name">{item.name}</span>
                <span className="qty">
                  <button onClick={() => onDecrease(item.id)} aria-label={'One fewer ' + item.name}>−</button>
                  <span aria-label={'Quantity of ' + item.name}>{item.qty}</span>
                  <button onClick={() => onIncrease(item.id)} aria-label={'One more ' + item.name}>+</button>
                </span>
                <span className="line-price">{won(item.price * item.qty)}</span>
              </li>
            ))}
          </ul>
        )}
      <p className="total" aria-live="polite"><span>Total</span><span>{won(total)}</span></p>
    </aside>
  );
}

function ShoppingCart({ products }) {
  const [cart, setCart] = useState([]);

  const isInCart = id => cart.some(item => item.id === id);
  const add = product => setCart(prev => [...prev, { ...product, qty: 1 }]);
  const remove = id => setCart(prev => prev.filter(item => item.id !== id));
  const increase = id => setCart(prev => prev.map(item =>
    item.id === id ? { ...item, qty: item.qty + 1 } : item
  ));
  const decrease = id => setCart(prev => prev
    .map(item => item.id === id ? { ...item, qty: item.qty - 1 } : item)
    .filter(item => item.qty > 0)
  );

  return (
    <div className="shop with-cart">
      <ProductList products={products} isInCart={isInCart}
                   onAdd={add} onRemove={remove} />
      <CartPanel cart={cart} onIncrease={increase} onDecrease={decrease} />
    </div>
  );
}

Same behaviour as section 4, split into components. ShoppingCart owns the state; ProductList and CartPanel only receive props and call back. This is lifting state up.
