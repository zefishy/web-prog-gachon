e.preventDefault

Stop the reload, then handle the submission yourself.

Submit without an @ and the form refuses; add one and it reports success. The page never navigates.

The snippet

const form = document.querySelector('#login-form');

form.addEventListener('submit', (e) => {
  e.preventDefault();            // stop the page reload

  const email = document.querySelector('#email').value;
  if (!email.includes('@')) {
    showError('Please enter a valid email.');
  } else {
    loginUser(email);
  }
});
